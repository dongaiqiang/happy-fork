## 一、企业版概述

### 1.1 企业版定位

面向企业客户的 HelloVibe 解决方案，提供：
- **私有化部署**：支持 VPC 和本地化部署
- **团队管理**：成员管理、角色权限、用量监控
- **安全合规**：审计日志、数据隔离、SSO 单点登录
- **服务保障**：SLA 99.9%、专属支持、定制培训

### 1.2 目标客户

| 客户类型 | 规模 | 核心需求 | 定价 |
|---------|------|---------|------|
| 中小企业 | 10-50 人 | SaaS 公有云、基础管理 | ¥999/月起 |
| 中大型公司 | 50-200 人 | VPC 部署、SSO、审计日志 | ¥4999/月起 |
| 大型企业 | 200 人+ | 本地部署、定制开发 | ¥19999 元/月起 |

## 四、API 设计

### 4.1 企业版 API 路由

**基础路径：** `/api/enterprise`

| 接口 | 方法 | 说明 |
|------|------|------|
| `/dashboard` | GET | 获取 Dashboard 概览数据 |
| `/members` | GET | 获取成员列表 |
| `/members` | POST | 邀请新成员 |
| `/members/:id` | DELETE | 移除成员 |
| `/members/:id/role` | PUT | 修改成员角色 |
| `/usage` | GET | 获取用量统计 |
| `/usage/export` | GET | 导出用量数据 |
| `/audit-logs` | GET | 获取审计日志 |
| `/audit-logs/export` | GET | 导出审计日志 |
| `/billing` | GET | 获取账单列表 |
| `/billing/:id/pay` | POST | 支付账单 |
| `/billing/:id/invoice` | GET | 下载发票 |
| `/sso/configs` | GET | 获取 SSO 配置列表 |
| `/sso/configs` | POST | 创建 SSO 配置 |
| `/sso/configs/:id` | PUT | 更新 SSO 配置 |
| `/sso/configs/:id` | DELETE | 删除 SSO 配置 |

## 五、数据模型

### 5.1 企业版相关 Schema

```prisma
// 企业组织
model Organization {
    id          String   @id @default(cuid())
    name        String
    tier        String   // "team" | "enterprise"
    plan        String   // "标准版" | "专业版" | "旗舰版"
    deployment  String   // "saas" | "vpc" | "on-premise"
    settings    Json?
    createdAt   DateTime @default(now())
    updatedAt   DateTime @updatedAt
    
    members     OrganizationMember[]
    subscriptions Subscription[]
    auditLogs   AuditLog[]
}

// 企业成员
model OrganizationMember {
    id             String       @id @default(cuid())
    organizationId String
    organization   Organization @relation(fields: [organizationId], references: [id])
    accountId      String
    account        Account      @relation(fields: [accountId], references: [id])
    role           String       // "admin" | "member" | "readonly"
    status         String       // "active" | "invited" | "suspended"
    invitedAt      DateTime     @default(now())
    joinedAt       DateTime?
    createdAt      DateTime     @default(now())
    updatedAt      DateTime     @updatedAt
    
    @@unique([organizationId, accountId])
}

// 审计日志
model AuditLog {
    id             String       @id @default(cuid())
    organizationId String
    organization   Organization @relation(fields: [organizationId], references: [id])
    accountId      String
    account        Account      @relation(fields: [accountId], references: [id])
    action         String       // "login", "api_call", "export", etc.
    resource       String?
    details        Json?
    ipAddress      String?
    userAgent      String?
    createdAt      DateTime     @default(now())
    
    @@index([organizationId, createdAt])
    @@index([accountId, createdAt])
}

// 订阅
model Subscription {
    id             String       @id @default(cuid())
    organizationId String
    organization   Organization @relation(fields: [organizationId], references: [id])
    planId         String
    status         String       // "active" | "cancelled" | "expired"
    startDate      DateTime     @default(now())
    endDate        DateTime?
    amount         Int
    currency       String       @default("CNY")
    billingCycle   String       // "monthly" | "annual"
    createdAt      DateTime     @default(now())
    updatedAt      DateTime     @updatedAt
    
    invoices       Invoice[]
}

// 发票
model Invoice {
    id             String       @id @default(cuid())
    subscriptionId String
    subscription   Subscription @relation(fields: [subscriptionId], references: [id])
    amount         Int
    status         String       // "pending" | "paid" | "overdue"
    dueDate        DateTime
    paidAt         DateTime?
    invoiceNumber  String       @unique
    downloadUrl    String?
    createdAt      DateTime     @default(now())
    
    @@index([subscriptionId])
}
```

### 6.3 权限控制

```typescript
// 角色权限定义
export const ROLE_PERMISSIONS = {
    admin: [
        'dashboard:view',
        'members:view',
        'members:invite',
        'members:remove',
        'members:edit-role',
        'usage:view',
        'usage:export',
        'audit-logs:view',
        'audit-logs:export',
        'billing:view',
        'billing:pay',
        'sso:manage',
        'settings:manage',
    ],
    member: [
        'dashboard:view',
        'usage:view',
        'audit-logs:view',
        'billing:view',
    ],
    readonly: [
        'dashboard:view',
    ],
};

// 权限检查 Hook
export function usePermission(permission: string) {
    const { currentRole } = useOrganization();
    const permissions = ROLE_PERMISSIONS[currentRole] || [];
    return permissions.includes(permission);
}
```
